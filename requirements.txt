# job-apply-os -- top-level requirements aggregator.

# Every service under services/*/ has its own requirements.txt (they're
# independently deployable). This file just pulls all of them together so
# a fresh clone can do one install instead of four:
#
#   python -m venv .venv
#   source .venv/bin/activate        # Windows/WSL: same command inside WSL bash
#   pip install -r requirements.txt
#
# `services/browser-controller/requirements.txt` also documents two
# system binaries (poppler, tesseract) that pip cannot install -- see
# README.md "OCR binaries" section for the Windows/WSL install commands.

-r services/browser-controller/requirements.txt

-r services/profile-ingestion/requirements.txt
